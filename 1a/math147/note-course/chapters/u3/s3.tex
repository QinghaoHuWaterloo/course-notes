\subsubsection{Euler's Number}
Consider the sequence 
\begin{equation*}
    x_n = \paren{1 + \frac{1}{n }}^n \qquad \forall n \in \N. 
\end{equation*}

\begin{theorem}
    [Binomial Theorem] 
    $$(1 + x )^n = 1 + nx + \frac{n(n - 1)}{2!}x^2 + \cdots$$
\end{theorem}
Thus, 
\begin{equation*}
    x_n = 1 + n \cdot \frac{1}{n} + n \cdot \frac{n - 1}{2!} \paren{\frac{1}{n^2}} + \cdots + \frac{n (n - 1) \cdots 1}{n!} \frac{1}{n^n}
\end{equation*}
\begin{equation*}
    = 1 + 1 + \frac{1}{2!} \frac{n }{n } \frac{n - 1}{n } + \frac{1}{3!} \frac{n }{n } \frac{n - 1}{n } \frac{n - 2}{n }+ \cdots + \frac{\paren{n - (n - 1)}}{n}
\end{equation*}
\begin{equation*}
    1 + 1 + \frac{1}{2!}\paren{1 - \frac{1}{n }} + \frac{1}{3!}\paren{1 - \frac{1}{n }}\paren{1 - \frac{2}{n }} + \cdots + \text{kiro, complete me! OH MY GOSH}
\end{equation*}
\begin{equation*}
    \leq 1 + \frac{1}{1!} + \frac{1}{2!} + \cdots + \frac{1}{n!}
\end{equation*}

Note that $\set{x_n }$ is decreasing. 
\begin{gather*}
    x_{n + 1} = 1 + \frac{1}{1!} + \frac{1}{2!}\paren{1 - \frac{1}{n + 1}} + \frac{1}{3!}\paren{1 - \frac{1}{n + 1}}\paren{1 - \frac{2}{n + 1}} + \cdots + \\\frac{1}{n!}\paren{1 - \frac{1}{n + 1}} \cdots \paren{1 - \frac{n - 1}{n + 1}}
    + \frac{1}{(n + 1)!}
\end{gather*}

Consider the new sequence
\begin{equation*}
    y_n = 1 + \frac{1}{1!} + \frac{1}{2!} + \cdots + \frac{1}{n!}
\end{equation*}
Note that $2 \leq y_n \leq 3$. 

Fact: $\frac{1}{n!} \leq \frac{1}{2^{n - 1}}$
\begin{equation*}
    \frac{1}{n!} = \frac{1}{1 \cdot 2 \cdots n} \leq \frac{1}{1} \cdot \frac{1}{2} \cdot \frac{1}{2} \cdots \frac{1}{2} = \frac{1}{2^{n - 1}}
\end{equation*}
\begin{equation*}
    y_n = 1 + \sum_{i = 1}^{n } \frac{1}{i!} \leq 1 + \sum_{i = 1}^{n } \frac{1}{2^{i - 1}} \leq 1 + \sum_{n = 1}^{\infty} (\frac{1}{2})^{i - 1}
\end{equation*}
\begin{equation*}
    = 1 + \frac{1}{1 - \frac{1}{2}} = 1 + 2 = 3. 
\end{equation*}
\begin{equation*}
    2 \leq x_n \leq y_n \leq 3
\end{equation*}

\subsubsection*{Subsequence}
\begin{definition}
    Let $\set{x_n }$ be a sequence of real numbers and let $n_1 < n_2 < n_3 < \cdots$ be a strictly increasing 
    sequence of natural numbers. Then 
    \begin{equation*}
        \set{x_{n_k }} = \set{x_{n_1}, x_{n_2}, x_{n_3}, \cdots}
    \end{equation*}

    is called a subsequence of $\set{x_n }$. 
\end{definition}
\begin{example}
    Consider the sequence $\set{x_n } = \set{1, -1, 1, -1, \cdots}$. 
\end{example}
In order to converge to 1, we take 
\begin{equation*}
    \set{x_{2k - 1}} = \set{1, 1, 1, \cdots}
\end{equation*}

Converge to $-1$
\begin{equation*}
    \set{x_{2_k}} = \set{-1, -1, -1, \cdots}
\end{equation*}

\begin{theorem}
    If a sequence $\set{x_n }$ of real numbers converges to a real number $x$, then any subsequence converges to $x$. 
\end{theorem}
\begin{proof}
    Let $\epsilon > 0$, $\exists n_\epsilon \in \N$ such that 
    \begin{equation*}
        n \geq n_\epsilon \implies \abs{x_n - x } < \epsilon. 
    \end{equation*}
    Now consider the subsequence 
    \begin{equation*}
        \set{x_{n_k}}
    \end{equation*}

    We claim: $k \leq n_k, \forall k \in \N$ HW: prove by induction. 

    Since $k \leq n_k$, we have $k + 1 \leq n_k + 1 \leq n_{k + 1}.$

    Thus, 
    \begin{equation*}
        n \geq n_\epsilon \implies n_k \geq k \geq n_\epsilon \implies \abs{x_{n_k} } < \epsilon
    \end{equation*}
\end{proof}
