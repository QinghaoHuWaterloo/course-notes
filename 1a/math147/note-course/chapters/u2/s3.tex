\subsection{The Algebraic and Order Properties of $\R$}
The real numbers $\R$ form a \emph{complete ordered field}. We build up this
description in three stages: first the field (algebraic) axioms, which govern
$+$ and $\cdot$; then the order axioms, which let us compare elements with
$<$; and finally the completeness axiom, which is what distinguishes $\R$ from
$\Q$ and is the source of all limiting behaviour in analysis. Each stage adds
structure on top of the previous one, and each is needed for a different
purpose: the field axioms let us compute, the order axioms let us compare, and
completeness lets us take limits.

\begin{definition}[Field]
    A \textbf{field} is a set $F$ equipped with two binary operations
    \[
        + : F \times F \to F,
        \qquad
        \cdot : F \times F \to F,
    \]
    satisfying the following axioms for all $a,b,c \in F$.

    \begin{enumerate}[label=\textbf{F\arabic*.}, leftmargin=2.5em]
        \item \textbf{Associativity of addition.}
        \[
            a + (b + c) = (a + b) + c.
        \]

        \item \textbf{Commutativity of addition.}
        \[
            a + b = b + a.
        \]

        \item \textbf{Additive identity.}
        There exists an element $0 \in F$ such that
        \[
            a + 0 = a.
        \]

        \item \textbf{Additive inverses.}
        For every $a \in F$, there exists an element $-a \in F$ such that
        \[
            a + (-a) = 0.
        \]

        \item \textbf{Associativity of multiplication.}
        \[
            a \cdot (b \cdot c) = (a \cdot b) \cdot c.
        \]

        \item \textbf{Commutativity of multiplication.}
        \[
            a \cdot b = b \cdot a.
        \]

        \item \textbf{Multiplicative identity.}
        There exists an element $1 \in F$, with $1 \neq 0$, such that
        \[
            a \cdot 1 = a.
        \]

        \item \textbf{Multiplicative inverses.}
        For every $a \in F$ with $a \neq 0$, there exists an element $a^{-1} \in F$ such that
        \[
            a \cdot a^{-1} = 1.
        \]

        \item \textbf{Distributive law.}
        \[
            a \cdot (b + c) = a \cdot b + a \cdot c.
        \]
    \end{enumerate}
\end{definition}

\begin{remark}
    The rational numbers $\Q$ also form a field: all nine axioms F1--F9 hold
    for the usual $+$ and $\cdot$ on fractions. So being a field alone does not
    distinguish $\R$ from $\Q$; the field axioms only capture the algebra of
    $+$ and $\cdot$, not any notion of size, order, or completeness.
\end{remark}

\begin{remark}
    In practice, F1--F9 are the axioms that justify every "obvious" algebraic
    manipulation: moving terms across an equals sign uses additive inverses
    and associativity, cancelling a nonzero factor uses multiplicative
    inverses, and expanding a product uses the distributive law. The theorem
    below derives a few of the most basic consequences (uniqueness of the
    identities, and $a \cdot 0 = 0$) directly from the axioms, to illustrate
    that even these familiar facts are theorems, not axioms, once the field
    axioms are fixed.
\end{remark}

\begin{theorem}
    \begin{enumerate}[label = \alph*)]
        \item If $z, a \in \R$ with $z + a = a$, then $z = 0$.
        \item If $u \in \R$ and $b \neq 0$ with $u \cdot b = b$, then $u = 1$.
        \item If $a \in \R$, then $a \cdot 0 = 0$.
    \end{enumerate}
\end{theorem}

\begin{proof}
    \begin{enumerate}[label = \alph*)]
        \item
        \[
            z = z + 0 = z + (a - a) = (z + a) + (-a) = a + (-a) = 0.
        \]
        \item Multiplying $u \cdot b = b$ by $b^{-1}$,
        \[
            u = u \cdot 1 = u \cdot \paren{b \cdot b^{-1}} = (u \cdot b) \cdot b^{-1} = b \cdot b^{-1} = 1.
        \]
        \item
        \[
            a\cdot0=a\cdot(0+0)=a\cdot0+a\cdot0.
        \]
        Adding $-(a\cdot0)$ to both sides, and using associativity, gives
        $0=a\cdot0$.
    \end{enumerate}
\end{proof}

\begin{definition}[The Order Properties of $\R$]
    There is a nonempty subset $P$ of $\R$, called the set of \textbf{positive real
    numbers}, that satisfies the following properties.
    \begin{enumerate}[label=\textbf{O\arabic*.}, leftmargin=3em]
        \item \textbf{Trichotomy law.}
        For every $a \in \R$, exactly one of the following holds:
        \begin{equation*}
            a = 0, \qquad a \in P, \qquad -a \in P.
        \end{equation*}

        \item \textbf{Closure under addition.}
        If $a, b \in P$, then $a + b \in P$.

        \item \textbf{Closure under multiplication.}
        If $a, b \in P$, then $a \cdot b \in P$.
    \end{enumerate}
\end{definition}

\begin{remark}
    O1 (trichotomy) is what makes $P$ usable for case-splitting arguments:
    given any real number, we may always split into the three exhaustive,
    mutually exclusive cases $a=0$, $a\in P$, $-a\in P$, and handle each case
    separately. O2 and O3 then guarantee that this splitting behaves well
    under the field operations, so that positivity is preserved by the
    algebra we already know how to do. This is exactly the pattern used below
    in the proof that $a\neq0 \implies a^2>0$.
\end{remark}

\begin{definition}[Order relation]
    Let $a, b \in \R$.
    \begin{enumerate}
        \item If $a - b \in P$, then we write $a > b$ or $b < a$.
        \item If $a - b \in P \cup \set{0}$, then we write $a \geq b$ or $b \leq a$.
    \end{enumerate}
\end{definition}

\begin{remark}
    This definition reduces every order statement to a statement about
    membership in $P$: to show $a > b$, it suffices to show $a - b \in P$.
    All order manipulations later in the course (adding the same quantity to
    both sides, multiplying by a positive quantity, etc.) ultimately trace
    back to this definition together with O1--O3.
\end{remark}

\begin{remark}
    $\Q$ is also ordered: the positive rationals $P \cap \Q$ satisfy O1--O3.
    So, just as with the field axioms, being an ordered field still does not
    distinguish $\R$ from $\Q$ -- we are still missing something. That
    missing ingredient is completeness, which $\Q$ does \emph{not} have (e.g.\
    $\set{q \in \Q : q^2 < 2}$ is bounded above in $\Q$ but has no least upper
    bound in $\Q$). This is precisely why completeness must be singled out as
    a separate axiom rather than derived from the field and order axioms.
\end{remark}

\begin{definition}[Completeness Axiom]
    If $A \subseteq \R$ is nonempty and bounded above, then $A$ has a least upper
    bound in $\R$.
\end{definition}

\begin{remark}
    This is only a preview statement: the notions of upper bound and least
    upper bound (supremum), and the full consequences of this axiom, are
    developed in \S 5. For now it suffices to record that this is the axiom
    that makes $\R$ complete, and hence suitable for limits, while $\Q$ is
    not.
\end{remark}

\begin{example}
    Prove that if $a \in \R$ and $a \neq 0$, then $a^2 > 0$.
\end{example}

\begin{proof}
    Since $a \neq 0$, the trichotomy law gives two cases.

    \textbf{Case 1: $a \in P$.} By closure under multiplication,
    \begin{equation*}
        a^2 = a \cdot a \in P, \qquad\text{so } a^2 > 0.
    \end{equation*}

    \textbf{Case 2: $a \notin P$.} Since $a \neq 0$, trichotomy rules out $a=0$,
    so it must be that $-a \in P$. By closure under multiplication,
    $(-a)(-a) \in P$. It remains to show that $a^2 = (-a)(-a)$, which we do by
    starting from $a + (-a) = 0$ and multiplying both sides by $(-a)$:
    \begin{align*}
        a + (-a) = 0
            &\implies (-a)\paren{a + (-a)} = (-a) \cdot 0 = 0 \\
            &\implies (-a)(a) + (-a)(-a) = 0 \\
            &\implies (-a)(-a) = -\paren{(-a)a} = -(-a^2) = a^2.
    \end{align*}
    Hence $a^2 = (-a)(-a) \in P$, so $a^2 > 0$. Since the two cases $a \in P$
    and $-a \in P$ are exhaustive (trichotomy, together with $a \neq 0$), we
    conclude $a^2 > 0$ in either case.
\end{proof}

\subsubsection{The Absolute Value}
\begin{definition}
    Given $a \in \R$, we define the \textbf{absolute value} $\abs{a}$ by
    \[
        \abs{a} =
        \begin{cases}
            a, & a \geq 0, \\
            -a, & a < 0.
        \end{cases}
    \]
\end{definition}

\begin{remark}
    Geometrically, $\abs{a}$ is the distance from $a$ to $0$ on the number
    line, and more generally $\abs{a-b}$ is the distance between $a$ and $b$.
    This interpretation is what makes the theorem and proposition below --
    especially the triangle inequality -- intuitive: they are just algebraic
    statements about distances.
\end{remark}

\begin{theorem}
    \begin{enumerate} [label = (\alph*)]
        \item $\abs{ab} = \abs{a}\abs{b}$ for all $a, b \in \R$.
        \item $\abs{a}^2 = a^2$ for all $a \in \R$.
        \item For $a \in \R$ and $c>0$,
        \[\abs{a} \leq c \iff -c \leq a \leq c.\]
        \item For every $a \in \R$,
        \[-\abs{a} \leq a \leq \abs{a}.\]

    \end{enumerate}

\end{theorem}

\begin{remark}[Proof idea]
    Parts (a), (b), (d) follow directly by splitting into the cases $a\geq0$
    and $a<0$ (and similarly for $b$) and unwinding the definition of
    $\abs{\cdot}$ in each case -- the same case-split strategy as in the
    $a^2>0$ example above. Part (c) is the most important part for later use:
    it converts a single inequality on $\abs{a}$ into a pair of ordinary
    inequalities on $a$, and its proof again splits on the sign of $a$.
\end{remark}

\begin{proof}
    The definition immediately gives $\abs{-a}=\abs{a}$: if $a\geq0$,
    then $-a\leq0$, and if $a<0$, then $-a>0$. It also gives
    \begin{equation*}
        -\abs{a}\leq a\leq\abs{a},
    \end{equation*}
    proving (d). In the two cases $a\geq0$ and $a<0$, respectively,
    $\abs{a}^2=a^2$ is immediate; this proves (b).

    For (a), there are four sign combinations. If $a,b\geq0$, then
    $ab\geq0$, so $\abs{ab}=ab=\abs a\abs b$. If $a\geq0>b$, then
    $ab\leq0$, so $\abs{ab}=-ab=a(-b)=\abs a\abs b$. The case
    $b\geq0>a$ is the same with the roles reversed. Finally, if $a,b<0$,
    then $ab>0$, and
    \begin{equation*}
        \abs{ab}=ab=(-a)(-b)=\abs a\abs b.
    \end{equation*}

    Finally let $c>0$. Suppose first that $\abs a\leq c$. If $a\geq0$,
    then $\abs a=a$, so $a\leq c$; also $-c<0\leq a$. If $a<0$, then
    $\abs a=-a$, so $-a\leq c$, equivalently $-c\leq a$; also $a<0<c$.
    Hence $-c\leq a\leq c$ in either case.

    Conversely, suppose $-c\leq a\leq c$. If $a\geq0$, then
    $\abs a=a\leq c$. If $a<0$, multiplying $-c\leq a$ by $-1$ reverses
    the inequality and gives $-a\leq c$, hence $\abs a\leq c$. This proves (c).
\end{proof}

\begin{remark}
    Part (c) is the key tool for turning an absolute-value inequality into a
    two-sided inequality (and vice versa); it will be used constantly in the
    rest of the course whenever a bound of the form $\abs{x}\leq c$ needs to
    be unpacked into $-c\leq x\leq c$, or vice versa. The triangle inequality
    below is the first major illustration of this.
\end{remark}

\begin{proposition}
    [Triangle Inequality]
    If $a, b \in \R$, then $\abs{a + b } \leq \abs{a} + \abs{b}$.
\end{proposition}

\begin{remark}[Proof idea]
    From part (d) of the preceding theorem we already have the two-sided
    bounds $-\abs{a}\leq a\leq\abs{a}$ and $-\abs{b}\leq b\leq\abs{b}$. Adding
    these two chains of inequalities termwise gives a two-sided bound on
    $a+b$ directly in terms of $\abs{a}+\abs{b}$. This is exactly the
    situation covered by part (c) of the same theorem, applied with $x=a+b$
    and $c=\abs{a}+\abs{b}$: a two-sided bound $-c\leq x\leq c$ converts back
    into the single inequality $\abs{x}\leq c$, which is precisely the
    triangle inequality we want.
\end{remark}

\begin{proof}
    Let $a, b \in \R$. By part (d) of the preceding theorem,
    \begin{align*}
        -\abs{a} \leq a \leq \abs{a}, \\
        -\abs{b} \leq b \leq \abs{b}.
    \end{align*}
    Adding these two chains of inequalities termwise gives
    \begin{align*}
        &-\abs{a} - \abs{b} \leq a + b \leq \abs{a} + \abs{b} \\
        &\iff - \paren{\abs{a} + \abs{b}} \leq a + b \leq \abs{a} + \abs{b}.
    \end{align*}

    If $\abs{a}+\abs{b}=0$, then the displayed bounds force $a+b=0$, so the
    claim is immediate. Otherwise $\abs{a}+\abs{b}>0$, and applying part (c)
    of the preceding theorem with $x = a+b$ and $c = \abs{a}+\abs{b}$ converts
    this two-sided bound back into the single inequality
    \begin{equation*}
        \abs{a + b} \leq \abs{a} + \abs{b}.
    \end{equation*}
\end{proof}

\begin{center}
    \fbox{
    \begin{minipage}{0.9\linewidth}
        \textbf{Key mechanism.} The bound $-\abs{a}\leq a\leq\abs{a}$ (part (d))
        added to its counterpart for $b$, combined with part (c)
        ($\abs{x}\leq c \iff -c\leq x\leq c$) applied to $x=a+b$,
        $c=\abs{a}+\abs{b}$, yields the triangle inequality
        $\abs{a+b}\leq\abs{a}+\abs{b}$.
    \end{minipage}
    }
\end{center}

\begin{remark}
    Interpreted via distance, the triangle inequality says that distance is
    subadditive: the direct distance from $0$ to $a+b$ never exceeds the sum
    of the distances from $0$ to $a$ and from $0$ to $b$.
\end{remark}

\begin{corollary}
    \begin{enumerate}
        \item $\abs{\abs{a} - \abs{b}} \leq \abs{a - b}.$
        \item $\abs{a - b} \leq \abs{a} + \abs{b}$
    \end{enumerate}
\end{corollary}

\begin{proof}[Proof of (1)]
    Let $a, b \in \R$. The idea is to write $a$ as $(a-b)+b$, so that the
    triangle inequality just proved applies directly to this sum and produces
    a bound involving $\abs{a-b}$:
    \begin{align*}
        a = (a - b) + b &\implies \abs{a} = \abs{(a - b) + b} \leq \abs{a - b} + \abs{b} \\
        &\implies \abs{a} - \abs{b} \leq \abs{a - b}.
    \end{align*}
    Symmetrically, writing $b$ as $(b-a)+a$ and applying the triangle
    inequality the same way gives
    \begin{align*}
        b = (b - a) + a &\implies \abs{b} = \abs{(b - a) + a} \leq \abs{b - a} + \abs{a} \\
        &\implies \abs{b} - \abs{a} \leq \abs{b-a}
            = \abs{-(a-b)}=\abs{a-b} \\
        &\implies \abs{a} - \abs{b} \geq - \abs{a - b}.
    \end{align*}

    Combining the two bounds obtained from these two decompositions,
    \begin{equation*}
        -\abs{a - b} \leq \abs{a} - \abs{b} \leq \abs{a - b}.
    \end{equation*}

    By part (c) of the absolute value theorem, applied with $x=\abs{a}-\abs{b}$
    and $c=\abs{a-b}$, this two-sided bound is equivalent to
    \begin{equation*}
        \abs{\abs{a} - \abs{b}} \leq \abs{a - b}.
    \end{equation*}
\end{proof}

\begin{proof}[Proof of (2)]
    Let $a, b \in \R$. Writing $a-b$ as $a+(-b)$ reduces this to a direct
    application of the triangle inequality, using $\abs{-b}=\abs{b}$:
    \begin{align*}
        \abs{a - b} = \abs{a + \paren{-b}} \leq \abs{a} + \abs{-b}
        = \abs{a} + \abs{b}.
    \end{align*}
\end{proof}
