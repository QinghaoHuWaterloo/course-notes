\subsection{The Completeness Property of $\R$}
\begin{axiom}[Completeness Property]
    Every nonempty set of real numbers that has an upper bound also has a supremum in $\R$.
\end{axiom}
\begin{corollary}
    Every nonempty set of real numbers that has a lower bound also has an infimum in $\R$.
\end{corollary}
\begin{proof}
    Let $S \subseteq \R$ be nonempty with a lower bound $w$, that is, $w \leq s$ for all $s \in S$. As a consequence, $-w \geq -s$ for all $s \in S$.

    Define $S' = \set{-s : s \in S}$. Then $S'$ is nonempty, and
    \begin{equation*}
        -w \geq -s, \ \forall s \in S \implies -w \text{ is an upper bound for } S'.
    \end{equation*}

    By the Completeness Axiom, $S'$ has a supremum. Say
    \begin{equation*}
        u := \sup S' \in \R.
    \end{equation*}

    We claim that $-u = \inf S$.
    \begin{itemize}
        \item \textbf{Lower bound:} Since $u = \sup S' \geq -s$ for all $s \in S$, we get $-u \leq s$ for all $s \in S$. So $-u$ is a lower bound for $S$.
        \item \textbf{Greatest lower bound:} Let $w$ be any lower bound for $S$. As shown above, $-w$ is then an upper bound for $S'$, so $u = \sup S' \leq -w$, which gives $w \leq -u$.
    \end{itemize}
    Thus $-u$ is a lower bound for $S$ that is at least as large as every other lower bound, so $-u = \inf S$.
\end{proof}

\begin{example}
    \begin{equation*}
        S = \set{\paren{1 + \frac{1}{n }}^n : n \in \N}
    \end{equation*}
    
    For every $n\in\N$, the binomial theorem gives
    \begin{equation*}
        \paren{1+\frac1n}^n
        =1+\binom n1\frac1n+\sum_{k=2}^n\binom nk\frac1{n^k}>1.
    \end{equation*}
    Indeed, for $0\leq k\leq n$,
    \begin{equation*}
        \binom nk=\frac{n(n-1)\cdots(n-k+1)}{k!}\leq\frac{n^k}{k!}.
    \end{equation*}
    For $k\geq2$, we have
    $k!\geq2^{k-1}$, because each factor from $2$ through $k$ is at least $2$. Therefore
    \begin{equation*}
        \paren{1+\frac1n}^n
        \leq\sum_{k=0}^n\frac1{k!}
        < 1+1+\sum_{k=2}^{\infty}\frac1{2^{k-1}}=3.
    \end{equation*}
    Thus $1<\paren{1+1/n}^n<3$ for all $n\in\N$.
\end{example}

\subsubsection{The Archimedean Property}
We now prove that $\N$ is not bounded above in $\R$. The proof combines two ingredients:
\begin{itemize}
    \item Completeness of $\R$.
    \item The inductive property of $\N$ (that $\N$ is closed under $+1$).
\end{itemize}

\begin{definition}
    [Archimedean Property]
    If $x \in \R$, then there exists $n_x \in \N$ such that $x \leq n_x$. 
\end{definition}
\begin{proof}
    Suppose, for contradiction, that $\exists x \in \R$ such that $x > n$ for all $n \in \N$.

    Then $x$ is an upper bound for $\N$. Since $1 \in \N$, $\N \neq \emptyset$, so by the Completeness Axiom, $\N$ has a supremum in $\R$.

    Say $u = \sup \N$. Since
    \begin{equation*}
        u - 1 < u,
    \end{equation*}
    $u - 1$ is not an upper bound for $\N$. Hence
    \begin{equation*}
        \exists m \in \N \text{ such that } u - 1 < m,
    \end{equation*}
    which gives
    \begin{equation*}
        u < m + 1.
    \end{equation*}
    But $m + 1 \in \N$ (since $\N$ is closed under $+1$), so $m + 1 \leq u$, contradicting $u < m+1$.

    Therefore no such $x$ exists; that is, for every $x \in \R$ there exists $n_x \in \N$ with $x \leq n_x$.
\end{proof}

\begin{proposition}
    There exists a real number $x$ such that $x^2 = 2$.
\end{proposition}
\begin{proof}
    Define
    \begin{equation*}
        S = \set{t \in \R : t \geq 0, t^2 < 2}.
    \end{equation*}
    We have $1 \in S$, so $S \neq \emptyset$, and $2$ is an upper bound for $S$: if $t\in S$ and $t>2$, then $t^2>4>2$, a contradiction. By the Completeness Axiom, $S$ has a supremum in $\R$.

    Let $x = \sup S$. Note $x \geq 1 > 0$. We show $x^2 = 2$ by ruling out $x^2 < 2$ and $x^2 > 2$.

    \textbf{Case 1: $x^2 < 2$.} We find $n \in \N$ large enough that $x + \frac{1}{n} \in S$, contradicting that $x$ is an upper bound for $S$. Since $x \geq 1$ and $n\geq1$,
    \begin{equation*}
        \paren{x + \frac{1}{n}}^2 = x^2 + \frac{2x}{n} + \frac{1}{n^2} \leq x^2 + \frac{2x}{n} + \frac{1}{n} = x^2 + \frac{2x+1}{n}.
    \end{equation*}
    Since $x^2 < 2$, the Archimedean Property lets us choose $n\in\N$ large enough that
    \begin{equation*}
        \frac{2x+1}{n} < 2 - x^2.
    \end{equation*}
    Then
    \begin{equation*}
        \paren{x + \frac{1}{n}}^2 < x^2 + (2 - x^2) = 2,
    \end{equation*}
    so $x + \frac{1}{n} \in S$. But $x + \frac{1}{n} > x = \sup S$, a contradiction. Hence $x^2 \geq 2$.

    \textbf{Case 2: $x^2 > 2$.} We find $n \in \N$ large enough that $x - \frac{1}{n}$ is an upper bound for $S$ smaller than $x$, contradicting that $x = \sup S$. Since
    \begin{equation*}
        \paren{x - \frac{1}{n}}^2 = x^2 - \frac{2x}{n} + \frac{1}{n^2} > x^2 - \frac{2x}{n},
    \end{equation*}
    the Archimedean Property lets us choose $n\in\N$ large enough that
    \begin{equation*}
        \frac{2x}{n} < x^2 - 2.
    \end{equation*}
    Then
    \begin{equation*}
        \paren{x - \frac{1}{n}}^2 > x^2 - (x^2 - 2) = 2.
    \end{equation*}
    So for any $t \in S$, we have $t^2 < 2 < \paren{x - \frac{1}{n}}^2$. Also $x-1/n\geq0$ because $x\geq1$ and $n\geq1$. Since $u\mapsto u^2$ is strictly increasing on $[0,\infty)$, this implies $t < x-\frac1n$. Thus $x - \frac{1}{n}$ is an upper bound for $S$ smaller than $x = \sup S$, a contradiction. Hence $x^2 \leq 2$.

    Combining both cases, $x^2 = 2$.
\end{proof}

\subsubsection{Dense of $\Q$}
\begin{remark}
    Consider the set
    \[  
        T=\{r\in\mathbb{Q}: r^2<2\}.
    \]
    Then \(T\neq\varnothing\), and \(T\) is bounded above in \(\mathbb{Q}\). For example,
    \[        0\in T
            \qquad\text{and}\qquad
            r<2 \quad \text{for every } r\in T.
    \]

    If \(T\) is regarded as a subset of \(\mathbb{R}\), then by the completeness of
    \(\mathbb{R}\), the supremum
    \[
        y=\sup_{\mathbb{R}} T
    \]
    exists.

    We can prove that
    \[
        y^2=2.
    \]
    Hence
    \[
        y=\sqrt{2}.
    \]
    Since
    \[
        \sqrt{2}\notin\mathbb{Q},
    \]
    it follows that \(T\) has no supremum in \(\mathbb{Q}\).

    Therefore, there exists a nonempty subset \(T\subset\mathbb{Q}\) which is
    bounded above in \(\mathbb{Q}\), but which has no least upper bound in
    \(\mathbb{Q}\). Thus \(\mathbb{Q}\) is not complete.
\end{remark}

\begin{theorem}
    If $x$ and $y$ are any real numbers with $x < y$, then there exists a rational number $r \in \Q$ such that $x < r < y$. 
\end{theorem}
\begin{proof}
    Suppose $x$ and $y$ are arbitrary real numbers where $x < y$. We have  
    \begin{equation*}
        x < y \iff y - x > 0. 
    \end{equation*}

    According to the Archimedean Property, it follows that 
    \begin{equation*}
        \paren{\exists n \in \N}\frac{1}{n } < y - x. \iff yn - xn > 1 \iff yn > 1 + xn. 
    \end{equation*}

    
\end{proof}

\begin{corollary}
    If $x$ and $y$ are any real numbers with $x < y$, then there exists an irrational number $z$ such that $x < r < y$. 
\end{corollary}

\begin{corollary}
    Let $x, y \in \R$, with $x < y$. Then there exists infinitely many rationals between $x$ and $y$. 
\end{corollary}

Are there more rational or irrational numbers? To answer this question, we need to know in what sense. 
\begin{definition}
    [Cardinality] Size of the set. 
\end{definition}

\begin{definition}
    Two sets have the same cardinality if there is a 1 - 1 (injective) and onto (surjective) function between sets. 
\end{definition}
\begin{example}
    \begin{enumerate}[label = (\alph*)]
        \item $\N$ and the set of even natural numbers. 
        \item $\N$ and $\Z$. 
    \end{enumerate}
\end{example}
\begin{definition}
    $S$ is called countable if $S$ has the same cardinality as $\N$. Otherwise, $S$ is called uncountable. 
\end{definition}
