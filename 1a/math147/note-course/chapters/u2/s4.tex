\subsection{Suprema and Infima}
Bounds make sense in any ordered set. The special feature of $\R$ is that every nonempty set bounded above has a supremum in $\R$; this need not hold in $\Q$.

\begin{definition}
    [Bounded Sets]\label{def:bounded-sets}
    Let $S$ be a nonempty subset of $\R$. The set $S$ is said to be\index{bounded set}\index{upper bound}\index{lower bound}
    \begin{enumerate}
        \item bounded above if there exists $u \in \R$ such that $x \leq u$ for all $x \in S$.
        \item bounded below if there exists $w \in \R$ such that $w \leq x$ for all $x \in S$.
        \item bounded if it is both bounded above and below.
        \item unbounded if it is not bounded.
    \end{enumerate}
\end{definition}

\begin{definition}
    [Supremum]\label{def:supremum}
    If there exists an upper bound $u_0$ of $S$ such that whenever $u$ is an upper bound for $S$
    we have $u_0 \leq u$, then $u_0$ is called the least upper bound, or the supremum, of $S$.\index{supremum}\index{least upper bound}
    \begin{equation*}
        u_0 := \sup S.
    \end{equation*}
\end{definition}

\begin{definition}
    [Infimum]\label{def:infimum}
    If there exists a lower bound $w_0$ of $S$ such that whenever $w$ is a lower bound for $S$
    we have $w_0 \geq w$, then $w_0$ is called the greatest lower bound, or the infimum, of $S$.\index{infimum}\index{greatest lower bound}
    \begin{equation*}
        w_0 := \inf S.
    \end{equation*}
\end{definition}


A set has at most one supremum. If $u_1$ and $u_2$ are both suprema of $S$, then $u_1$ is least and $u_2$ is an upper bound, so $u_1\leq u_2$; swapping roles gives $u_2\leq u_1$. Hence $u_1=u_2$, and the same argument with inequalities reversed shows the infimum is unique.

\subsubsection{The $\epsilon$-Characterization of the Supremum}
When does an upper bound $u$ fail to be the \emph{least} one? Exactly when some smaller number is still an upper bound --- that is, when there is a gap between $S$ and $u$. So $u$ is the supremum precisely when there is no gap: however small a margin $\epsilon>0$ we allow, $S$ still has a point within $\epsilon$ of $u$.


\mdfsetup{nobreak=true}
\begin{lemma}\label{lem:eps-characterization}
    An upper bound $u$ of a nonempty set $S$ in $\R$ is the supremum of $S$ if and only if\index{supremum!epsilon-characterization@$\epsilon$-characterization}
    \begin{equation*}
        (\forall \epsilon > 0)(\exists x_\epsilon \in S)\,[u - \epsilon < x_\epsilon].
    \end{equation*}
    \insymbols{\begin{aligned}
        &(\forall S\subseteq\R)(\forall u\in\R)\,\Bigl[\bigl(S\neq\emptyset\wedge(\forall x\in S)\,[x\leq u]\bigr)\\
        &\quad\Rightarrow\bigl(u=\sup S\iff(\forall\epsilon>0)(\exists x_\epsilon\in S)\,[u-\epsilon<x_\epsilon]\bigr)\Bigr]
    \end{aligned}}
\end{lemma}
\mdfsetup{nobreak=false}

\begin{proof}
    ($\Rightarrow$) Suppose $u=\sup S$. Let $\epsilon>0$. Since
    \begin{equation*}
        u-\epsilon<u,
    \end{equation*}
    the number $u-\epsilon$ cannot be an upper bound: otherwise it would be an upper bound smaller than $u$, contradicting that $u$ is the \emph{least} upper bound. Since $u - \epsilon$ is not an upper bound, by the definition of upper bound there is some $x_\epsilon\in S$ such that
    \begin{equation*}
        u-\epsilon<x_\epsilon\leq u.
    \end{equation*}

    ($\Leftarrow$) Conversely, suppose $u$ is an upper bound and that for every $\epsilon>0$ there exists $x_\epsilon\in S$ with $u-\epsilon<x_\epsilon$. If $u$ were not the least upper bound, there would be another upper bound $w<u$. Choose
    \begin{equation*}
        \epsilon=u-w>0.
    \end{equation*}
    The assumed condition then yields some $x_\epsilon \in S$ with
    \begin{equation*}
        w=u-\epsilon<x_\epsilon,
    \end{equation*}
    contradicting that $w$ is an upper bound (since $x_\epsilon \in S$ exceeds $w$). Hence no such $w$ exists, and $u=\sup S$.
\end{proof}


\begin{example}
    Consider the set $S = \set{x \in \R : 0 \leq x < 1}$. Then $\sup S = 1$ and $\inf S = 0$.
\end{example}

\begin{proof}[Proof that $\sup S=1$]
    Every $x\in S$ satisfies $x<1$, so $1$ is an upper bound. Let $u=\sup S$, which exists by completeness since $S$ is nonempty and bounded above. Because $1$ is an upper bound, $u\leq1$. Suppose $u\neq1$, so $u<1$. Choose
    \begin{equation*}
        \epsilon=\frac{1-u}{2}>0.
    \end{equation*}
    Then
    \begin{equation*}
        u<u+\epsilon=\frac{u+1}{2}<1.
    \end{equation*}
    Also $u\geq0$ since $0\in S$, so $u+\epsilon\geq0$. Thus $u+\epsilon\in S$, contradicting that $u$ is an upper bound for $S$. Therefore $u=1$, i.e.\ $\sup S=1$.
\end{proof}

\begin{proof}[Proof that $\inf S=0$]
    Since $x\geq0$ for all $x\in S$, $0$ is a lower bound. Let $w=\inf S$. Because $0$ is a lower bound, $w\geq0$. Suppose $w\neq0$, so $w>0$. But $0\in S$, and a lower bound $w$ must satisfy $w\leq0$, a contradiction. Therefore $w=0$, i.e.\ $\inf S=0$.
\end{proof}

\begin{remark}
    In this example, $S=[0,1)$ has $\sup S=1$ but no maximum, because $1\notin S$. Its infimum $0$ does belong to $S$, so $0$ is also its minimum. A supremum is the \emph{least upper bound}, whether or not it belongs to the set; it is a maximum only when it does belong to the set. Similarly, ``bounded above,'' ``bounded below,'' ``bounded,'' and ``unbounded'' are properties of the \emph{set} $S$, whereas $\sup S$ and $\inf S$ (when they exist) are specific \emph{real numbers} associated to $S$.
\end{remark}

\subsubsection*{In practice}
\begin{itemize}
    \item \textbf{Verifying $u=\sup S$.} Check two things: (i) $u$ is an upper bound for $S$; (ii) no smaller number is. Condition (ii) quantifies over all numbers below $u$ and is awkward directly; use the $\epsilon$-form instead: for each $\epsilon>0$, exhibit $x\in S$ with $u-\epsilon<x\leq u$.
    \item \textbf{Infimum.} The same recipe with inequalities reversed: a lower bound $w$, and for each $\epsilon>0$ some $x\in S$ with $w\leq x<w+\epsilon$.
\end{itemize}
