% Each problem starts a numbered section titled "Problem N".
\newcounter{problem}

\newenvironment{problem}[1][]{%
  \refstepcounter{problem}%
  \bigskip
  \noindent
  \begin{mdframed}[
    linewidth=0.6pt,
    linecolor=black!55,
    backgroundcolor=black!3,
    innertopmargin=6pt,
    innerbottommargin=6pt,
    innerleftmargin=8pt,
    innerrightmargin=8pt,
    roundcorner=3pt
  ]
  \textbf{Problem \theproblem}%
  \ifx\relax#1\relax\else\ \textbf{(#1)}\fi.\quad
}{%
  \end{mdframed}
}
