% "Boxed" style: full border + drop shadow, sharp corners.
% Used for theorem / lemma / definition.
\newcommand{\boxedNote}[3]{%
  \surroundwithmdframed[
    linecolor=#2, backgroundcolor=#3, linewidth=1.2pt,
    roundcorner=0pt,
    shadow=true, shadowsize=4pt, shadowcolor=black!40,
    skipabove=10pt, skipbelow=10pt,
    innertopmargin=7pt, innerbottommargin=7pt,
    innerleftmargin=9pt, innerrightmargin=9pt
  ]{#1}%
}

% "Sidebar" style: thick left bar only, tinted fill, no border or
% shadow. Used for proposition / corollary.
\newcommand{\sidebarNote}[3]{%
  \surroundwithmdframed[
    topline=false, bottomline=false, rightline=false, leftline=true,
    linecolor=#2, linewidth=3pt, backgroundcolor=#3,
    roundcorner=0pt,
    skipabove=8pt, skipbelow=8pt,
    innertopmargin=6pt, innerbottommargin=6pt,
    innerleftmargin=9pt, innerrightmargin=6pt
  ]{#1}%
}

\boxedNote{theorem}{thmAccent}{thmFill}
\boxedNote{lemma}{lemAccent}{lemFill}
\boxedNote{definition}{defAccent}{defFill}

\sidebarNote{proposition}{propAccent}{propFill}
\sidebarNote{corollary}{corAccent}{corFill}

\surroundwithmdframed[
  topline=false, bottomline=false, rightline=false, leftline=true,

  innertopmargin=3pt, innerbottommargin=3pt,
  innerleftmargin=8pt, innerrightmargin=4pt
]{example}

% Proof head: bold, non-italic, on its own line (matches the other
% note boxes instead of amsthm's italic "Proof." run-in default).

  \par\noindent{\bfseries #1}\par\nopagebreak\normalfont\noindent\ignorespaces
