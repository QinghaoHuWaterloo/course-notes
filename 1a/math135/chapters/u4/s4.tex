\subsection{Proving If and Only If Statements}
The statement $A\iff B$ means $(A\implies B)\wedge(B\implies A)$.
Prove both directions to establish the equivalence.

\begin{example}
    Prove that for all non-negative $x, y \in \R$,
    \begin{equation*}
        \frac{x+y}{2}=\sqrt{xy}\quad\iff\quad x=y.
    \end{equation*}
\end{example}
\begin{remark}[Proof idea]
    If $x=y$, evaluate both sides. In the other direction, square the equality
    and rearrange it into $(x-y)^2=0$.
\end{remark}
\begin{proof}
    Let $x,y\in\R$ be arbitrary and non-negative.

    $(\Leftarrow)$ Suppose $x=y$. Then
    \begin{equation*}
        \frac{x+y}{2}=y=\sqrt{y^2}=\sqrt{xy},
    \end{equation*}
    where $\sqrt{y^2}=y$ because $y\geq 0$.

    $(\Rightarrow)$ Suppose $\frac{x+y}{2}=\sqrt{xy}$. Squaring both sides and
    multiplying by $4$ gives
    \begin{equation*}
        x^2+2xy+y^2=4xy,
    \end{equation*}
    so $x^2-2xy+y^2=(x-y)^2=0$. Hence $x-y=0$, that is, $x=y$.
\end{proof}

\subsubsection*{In practice}
\begin{itemize}
    \item \textbf{Two directions.} Label the parts $(\Rightarrow)$ and
    $(\Leftarrow)$; in each, assume one side and derive the other. Proving
    only one direction proves an implication, not the equivalence.
\end{itemize}
