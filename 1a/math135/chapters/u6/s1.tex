\subsection{Principle of Mathematical Induction}
Induction proves a statement for every natural number at once: we check the
first case, then show that each case implies the next.

\begin{axiom}[The Principle of Mathematical Induction]
    Let $P(n)$ be an open sentence that depends on $n \in \N$. If
    \begin{enumerate}[label = (\arabic*)]
        \item $P(1)$ is true, and
        \item $\forall k \in \N,\ P(k) \implies P(k + 1)$,
    \end{enumerate}
    then $P(n)$ is true for all $n \in \N$.
\end{axiom}
Condition (2) is a universally quantified implication: it does not assert
$P(k)$, only that $P(k)$ would force $P(k+1)$. Together with (1), it yields
$P(1)\implies P(2)\implies P(3)\implies\cdots$.

\subsubsection{Proof by Mathematical Induction}
To prove a universally quantified statement $\forall n \in \N,\ P(n)$:
\begin{itemize}
    \item \textbf{Base case.} Prove $P(1)$.
    \item \textbf{Inductive step.} Let $k\in\N$ be arbitrary, assume $P(k)$
    (the \emph{inductive hypothesis}), and prove $P(k+1)$.
\end{itemize}

\begin{example}
    Prove that for all $n \in \N$,
    \begin{equation*}
        \sum_{i = 1}^{n} i\paren{i + 1} = \frac{1}{3}\, n \paren{n + 1} \paren{n + 2}.
    \end{equation*}
\end{example}
\begin{remark}[Proof idea]
    The sum up to $k+1$ is the sum up to $k$ plus one new term. Replace the
    first part using the inductive hypothesis, then factor out
    $(k+1)(k+2)$.
\end{remark}
\begin{proof}
    Let $P(n)$ be the statement above.

    \textbf{Base case.} $P(1)$ is true, since
    \begin{equation*}
        \sum_{i = 1}^{1} i\paren{i + 1} = 1 \cdot 2 = 2 = \frac{1}{3} \cdot 1 \cdot 2 \cdot 3.
    \end{equation*}

    \textbf{Inductive step.} Let $k$ be an arbitrary natural number and
    assume $P(k)$, that is,
    \begin{equation*}
        \sum_{i = 1}^{k} i \paren{i + 1} = \frac{1}{3}\, k\paren{k + 1}\paren{k + 2}.
    \end{equation*}
    We prove $P(k + 1)$. Splitting off the last term and using the inductive
    hypothesis,
    \begin{align*}
        \sum_{i = 1}^{k + 1} i \paren{i + 1}
        &= \sum_{i = 1}^{k} i\paren{i + 1} + \paren{k + 1}\paren{k + 2} \\
        &= \frac{1}{3}\, k\paren{k + 1}\paren{k + 2} + \paren{k + 1}\paren{k + 2} \\
        &= \paren{k + 1}\paren{k + 2}\paren{\frac{k}{3} + 1} \\
        &= \frac{1}{3} \paren{k + 1}\paren{k + 2}\paren{k + 3}.
    \end{align*}
    This is $P(k+1)$. By the Principle of Mathematical Induction, $P(n)$ is
    true for all $n \in \N$.
\end{proof}

\subsubsection{Starting at a Later Base Case}
If a statement is claimed only for integers $n\geq b$, the base case is
$n=b$ and the inductive step assumes $k\geq b$. The argument is the same:
the base case starts the chain at $b$, and each step moves it one further.

\begin{example}
    Prove that $n! > 2^n$ for every integer $n \geq 4$.
\end{example}
\begin{remark}[Proof idea]
    Going from $k$ to $k+1$ multiplies the left side by $k+1$ and the right
    side by $2$. Since $k+1\geq 2$, the left side grows at least as fast.
\end{remark}
\begin{proof}
    We use induction on $n$, starting at $n=4$.

    \textbf{Base case.} For $n = 4$, the left side is $4! = 24$ and the right
    side is $2^4 = 16$. Since $24 > 16$, the statement holds for $n = 4$.

    \textbf{Inductive step.} Let $k$ be an arbitrary integer with $k \geq 4$,
    and assume $k! > 2^k$. We prove $\paren{k + 1}! > 2^{k + 1}$.

    Multiplying the inductive hypothesis by $k+1>0$ gives
    \begin{equation*}
        \paren{k + 1}! = \paren{k + 1}k! > \paren{k + 1}2^k.
    \end{equation*}
    Since $k \geq 4$, we have $k + 1 \geq 5 \geq 2$, so
    \begin{equation*}
        \paren{k + 1}! > \paren{k + 1}2^k \geq 2 \cdot 2^k = 2^{k + 1}.
    \end{equation*}
    Hence the statement holds for $k+1$. By induction, $n! > 2^n$ for every
    integer $n \geq 4$.
\end{proof}
\begin{remark}
    The base case cannot be moved earlier: $3! = 6 < 8 = 2^3$. The inductive
    step alone proves nothing; it needs a true starting case.
\end{remark}

\subsubsection*{In practice}
\begin{itemize}
    \item \textbf{Induction template.} State $P(n)$; prove the base case;
    write ``Let $k$ be arbitrary and assume $P(k)$''; write out $P(k+1)$ as
    the goal; conclude ``by the Principle of Mathematical Induction''.
    \item \textbf{Sums.} Split off the last term,
    $\sum_{i=1}^{k+1}t_i=\sum_{i=1}^{k}t_i+t_{k+1}$, so the inductive
    hypothesis applies to the first part.
    \item \textbf{Inequalities.} Compare the factor each side gains from
    $k$ to $k+1$, and use the range of $k$ (here $k\geq4$) to compare them.
\end{itemize}
