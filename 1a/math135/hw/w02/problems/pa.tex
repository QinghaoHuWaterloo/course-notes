\section{Q5}
\begin{problem}
    Prove the following statement:
    \[
        \forall p,q,r\in\mathbb{R},
        \qquad
        2(p^2+13r^2)
        +q(5q+4p-10r)
        \geq 2pr.
    \]
\end{problem}

\begin{solution}
    Suppose $p, q, r$ are all arbitrary real numbers in $\R$. 

    According to the real numbers property, we have 
    \begin{equation*}
        \paren{p - r}^2 \geq 0 \quad (p + 2q)^2 \geq 0 \quad (q - 5r)^2 \geq 0,  
    \end{equation*}

    Adding all the inequilities together, we get 
    \begin{equation*}
        \paren{p - r}^2 + \paren{p + 2q}^2 + \paren{q - 5r}^2 \geq 0. 
    \end{equation*}

    Following from the square of binomial, 
    \begin{equation*}
         (p^2 - 2pr + r^2) + (p^2 + 4pq + 4q^2) + (q^2 - 10qr + 25r^2) \geq 0. 
    \end{equation*}

    Expanding the parenthesis, we have 
    \begin{equation*}
         p^2 - 2pr + r^2 + p^2 + 4pq + 4q^2 + q^2 - 10qr + 25r^2 \geq 0. 
    \end{equation*}

    Simplifying the inequility, this yields:
    \begin{equation*}
        2p^2 + 26r^2 + 5q^2 + 4pq - 10qr - 2pr \geq 0. 
    \end{equation*}
    
    Adding $2pr$ for both sides, we have 
    \begin{equation*}
        2p^2 + 26r^2 + 5q^2 + 4pq - 10qr \geq 2pr. 
    \end{equation*}

    Factoring $2$ out of $2p^2 + 26r^2$ and $q$ out of  $5q^2 + 4pq - 10qr$, we get 
    \[
        2(p^2+13r^2)
        +q(5q+4p-10r)
        \geq 2pr.
    \]

    Therefore, the original inequility is true. 
\end{solution}
