\begin{enumerate}[label = (\alph*)]
    \item \begin{gather*}
        \overline{G} = \set{3, 5, 7, 8, 10 } \\
        \overline{G} \cap H = \set{5, 7}. 
    \end{gather*}

    \item \begin{gather*}
        F \cup H = \set{1, 4, 5, 7 , 9, 10} \\
        G \cap \paren{F \cup H} = \set{1, 4, 9}
    \end{gather*}

    \item \begin{gather*}
        G \cup H = \set{1, 2, 4, 5, 6, 7, 9} \\
        \overline{\paren{G \cup H}} = \set{3, 8, 10}
    \end{gather*}

    \item \begin{gather*}
        \paren{F \cap H} = \set{4, 5, 9} \\
        \paren{F  \cap H } - G = \set{5}
    \end{gather*}
\end{enumerate}
